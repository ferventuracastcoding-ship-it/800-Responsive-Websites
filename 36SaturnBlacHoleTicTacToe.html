💻 MINIMAX 

function minimax(board, depth, isMaximizing) {

    if (gameOver(board)) {
        return score(board);
    }

    if (isMaximizing) {

        let bestScore = -Infinity;

        for (const move of emptyCells(board)) {

            board[move] = BLACK_HOLE;

            let score =
                minimax(
                    board,
                    depth + 1,
                    false
                );

            board[move] = null;

            bestScore =
                Math.max(
                    bestScore,
                    score
                );
        }

        return bestScore;
    }
}
