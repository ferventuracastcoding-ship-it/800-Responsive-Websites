15. Spiral




A spiral-like pattern generated using distance and angles.




JavaScript



for(let row=0;row<17;row++){

    for(let col=0;col<17;col++){

        let dx=col-8;
        let dy=row-8;

        let d=Math.sqrt(dx*dx+dy*dy);

        if(Math.floor(d)%2===0)
            result+="•";
    }
}
